3-QADAM: Subcategory jadvallarini to'ldirish (Har biriga 6 tadan = 102 ta)
Kategoriyalar ID raqamlari o'zgaruvchan bo'lishi mumkinligini hisobga olib, bazadagi mavjud kategoriyalarni siklda aylanib chiqamiz va xatosiz bog'laymiz.

SQL
DO $$
DECLARE
    cat_rec RECORD;
    sub_idx INT;
BEGIN
    -- Bazadagi bor barcha kategoriyalarni navbatma-navbat olamiz
    FOR cat_rec IN SELECT id, name FROM "Category" LOOP
        -- Har biri ichiga 6 tadan subkategoriya yaratamiz
        FOR sub_idx IN 1..6 LOOP
            INSERT INTO "Subcategory" (name, description, category_id, created_at, updated_at)
            VALUES (
                'Подкатегория ' || sub_idx || ' для [' || cat_rec.name || ']',
                'Описание подкатегории ' || sub_idx,
                cat_rec.id, -- To'g'ri bog'lanish
                NOW(),
                NOW()
            );
        END LOOP;
    END LOOP;
END $$;
4-QADAM: ChildCategory jadvallarini to'ldirish (Har biriga 10 tadan = 1020 ta)
Yaratilgan barcha subkategoriyalar IDlarini massivga yig'ib, xavfsiz va xatosiz tarzda har biriga 10 tadan quyi-kategoriya biriktiramiz.

SQL
DO $$
DECLARE
    sub_rec RECORD;
    child_idx INT;
BEGIN
    -- Yaratilgan 102 ta subkategoriyani aylanib chiqamiz
    FOR sub_rec IN SELECT id, name FROM "Subcategory" LOOP
        -- Har biriga 10 tadan child kategoriya
        FOR child_idx IN 1..10 LOOP
            INSERT INTO "ChildCategory" (name, description, subcategory_id, created_at, updated_at)
            VALUES (
                'Дочерняя категория ' || child_idx || ' (' || sub_rec.name || ')',
                'Описание дочерней категории',
                sub_rec.id, -- To'g'ri bog'lanish
                NOW(),
                NOW()
            );
        END LOOP;
    END LOOP;
END $$;
5-QADAM: Product jadvallarini to'ldirish (200 ta mahsulot)
Bu skript ChildCategory va Brand jadvallaridagi haqiqiy ID'larni yig'ib olib, ular o'rtasida 200 ta tayyor mahsulotni FOREIGN KEY cheklovlarini buzmasdan muvozanatli taqsimlaydi.

SQL
DO $$
DECLARE
    child_ids INT[] := ARRAY(SELECT id FROM "ChildCategory");
    brand_ids INT[] := ARRAY(SELECT id FROM "Brand");
    child_count INT := array_length(child_ids, 1);
    brand_count INT := array_length(brand_ids, 1);
    current_child_id INT;
    current_brand_id INT;
BEGIN
    FOR g IN 1..200 LOOP
        -- ID ketma-ketligini matematik qoldiq orqali aniqlash
        current_child_id := child_ids[((g - 1) % child_count) + 1];
        current_brand_id := brand_ids[((g - 1) % brand_count) + 1];

        INSERT INTO "Product" (
            name, description, price, image_url, stock_quantity, 
            child_category_id, brand_id, full_description, is_active, 
            discount, characteristics, created_at, updated_at
        )
        VALUES (
            'Электронный гаджет модель №' || g,
            'Краткое описание товара №' || g,
            (100 + (g * 10))::DOUBLE PRECISION,
            'product_' || g || '.png',
            15 + (g % 40),
            current_child_id, -- Bazada mavjud Child ID
            current_brand_id, -- Bazada mavjud Brand ID
            'Полное детальное описание продукта со всеми техническими подробностями для модели №' || g,
            TRUE,
            CASE WHEN g % 5 = 0 THEN 15.0 ELSE 0.0 END,
            '{"Цвет": "Черный", "Гарантия": "12 месяцев"}'::jsonb,
            NOW(),
            NOW()
        );
    END LOOP;
END $$;
